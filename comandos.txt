# ==========================================
# GUIA DE DEPLOY: VPS (UBUNTU)
# ==========================================

# 1. Atualizar a VPS e instalar o Docker
sudo apt update && sudo apt upgrade -y
sudo apt install -y docker.io docker-compose-v2

# 2. Habilitar o serviço do Docker para rodar na inicialização
sudo systemctl enable docker
sudo systemctl start docker

# 3. Mover/Clonar o projeto para a VPS
# (Se for por Github):
# git clone <url_do_seu_repo>
# cd dom-lux-brindes

# 4. Construir e subir todos os containers (Banco, Redis, Celery, Backend, Frontend)
sudo docker compose up -d --build

# 5. Criar as tabelas no Banco de Dados Postgres 
# (Roda as migrations dentro do container web)
sudo docker compose exec web python manage.py makemigrations
sudo docker compose exec web python manage.py migrate

# 6. Criar um usuário Administrador para o painel Django e para usar o sistema
sudo docker compose exec web python manage.py createsuperuser

# 7. Coletar os arquivos estáticos do admin do Django (Se necessário)
sudo docker compose exec web python manage.py collectstatic --noinput

# ==========================================
# OUTROS COMANDOS ÚTEIS PARA GERENCIAR
# ==========================================
# Ver os logs do backend se der erro:
# sudo docker compose logs -f web

# Ver os logs do Celery (background tasks):
# sudo docker compose logs -f celery

# Parar tudo:
# sudo docker compose down
